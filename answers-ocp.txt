QUESTION 12:
# Pre-Prequits
oc new-app --name=qed openshift/httpd

Answer:
oc project
oc get all
oc set env --from=secret/magic deployment/qed
oc set env deployment/qed –list  ## To check

cat qed.yaml 
apiVersion: apps/v1
kind: Deployment
metadata:
  name: qed
  namespace: math
spec:
  replicas: 1
  selector:
    matchLabels:
      app: qed
  template:
    metadata:
      labels:
        app: qed
    spec:
      containers:
      - name: web
        image: quay.io/fedora/python-39
        command: ["/bin/sh", "-c"]
        args:
          - >
            decoded=$(echo "${DECODER_RING}" | base64 --decode);
            echo "Welcome to the QED web application!" > /tmp/index.html &&
            echo "<br>Decrypted Decoder Ring: ${decoded}" >> /tmp/index.html &&
            cd /tmp &&
            python3 -m http.server 8080
        env:
        - name: DECODER_RING
          valueFrom:
            secretKeyRef:
              name: magic
              key: decoder_ring
        ports:
        - containerPort: 8080
---
apiVersion: v1
kind: Service
metadata:
  name: qed
  namespace: math
spec:
  selector:
    app: qed
  ports:
    - protocol: TCP
      port: 80
      targetPort: 8080
---
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: qed
  namespace: math
spec:
  to:
    kind: Service
    name: qed
  port:
    targetPort: 8080
  tls:
    termination: edge
